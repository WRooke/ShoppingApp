# ShoppingApp — dependencies
# Kept deliberately minimal.
#
# Pinned to exact versions (not just floors) so a fresh `pip install` on the NUC — which
# start.bat and scripts/update.py both do unattended — can't silently pick up a newer
# release than what was tested on the dev PC. Direct dependencies only; pip still resolves
# compatible transitive versions itself. Bump deliberately, not as a side effect of an
# unrelated change.

fastapi==0.141.1
uvicorn==0.52.4
SQLAlchemy==2.0.52
alembic==1.19.2  # DB migrations — bootstrapped Phase 3 Chunk 3.7a (see CLAUDE.md > Migrations)
python-dotenv==1.2.3

# Dev/test — see CLAUDE.md > Code Architecture & Maintainability. Added ahead of Phase 2
# so the tests/ convention exists from the first real business logic, not retrofitted later.
pytest==9.1.1
httpx==0.28.1  # required by fastapi.testclient.TestClient; also used directly in Phase 3
              # for recipe URL fetching (see services/capture_url.py)

# Phase 3 — recipe capture (AI). Google Gemini as of Phase 3.9 (was anthropic; see CLAUDE.md >
# AI Provider Migration). google-genai pulls google-auth + its own requests/websockets.
google-genai==2.22.0
beautifulsoup4==4.15.0  # HTML parsing for URL capture (services/capture_url.py)
python-multipart==0.0.32  # required by FastAPI to parse the photo-upload multipart form
brotli==1.2.0  # httpx's brotli decoder for URL capture — see services/capture_url.py; without
                # it, some bot-protected recipe sites (e.g. tasty.co) 406 the request outright
                # (their WAF treats a Chrome UA with no brotli support as non-browser), and if
                # they ever did serve br-encoded content anyway it would decode to mojibake

# Phase 5 — AnyList integration. The connector itself is native (httpx + a hand-rolled
# protobuf codec, see services/anylist_client.py, adapted from spike/). `keyring` is only
# for reading the AnyList credentials out of Windows Credential Manager when they're stored
# there rather than in .env — see CLAUDE.md > Security §2 (hybrid credential storage).
keyring==25.7.0
